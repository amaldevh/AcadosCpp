cmake_minimum_required(VERSION 3.18)
project(quadrotor_control_loop LANGUAGES CXX)

set(ACADOS_CPP_GENERATED_DIR
    "${CMAKE_CURRENT_SOURCE_DIR}/cpp_quadrotor_ocp"
    CACHE PATH "Directory generated by generate_cpp_ocp.py")

if(NOT EXISTS "${ACADOS_CPP_GENERATED_DIR}/CMakeLists.txt")
    message(FATAL_ERROR
        "No generated AcadosCpp project found at ${ACADOS_CPP_GENERATED_DIR}. "
        "Run quadrotor_model.py and generate_cpp_ocp.py first.")
endif()

add_subdirectory(
    "${ACADOS_CPP_GENERATED_DIR}"
    "${CMAKE_CURRENT_BINARY_DIR}/acados_cpp_generated")

add_executable(control_loop cpp_control_loop.cc)
target_compile_features(control_loop PRIVATE cxx_std_17)
target_link_libraries(control_loop PRIVATE AcadosCpp::ocp AcadosCpp::sim)

# Run from the examples directory so the CSV is written beside the scripts.
set_target_properties(control_loop PROPERTIES
    BUILD_RPATH "${ACADOS_CPP_GENERATED_DIR};${ACADOS_LIB_PATH}")
